import fs from 'node:fs';
import path from 'node:path';
import assert from 'node:assert/strict';
import { createHash } from 'node:crypto';
import sharp from 'sharp';
const ROOT=path.resolve(import.meta.dirname,'../../..'),D=path.relative(ROOT,import.meta.dirname),P=p=>path.resolve(ROOT,p),read=p=>fs.readFileSync(P(p)),json=p=>JSON.parse(read(p)),sha=b=>createHash('sha256').update(b).digest('hex');
const records=json(D+'/generation-records.json');assert.equal(records.records.length,40);assert.equal(records.remainingHeroIds.length,0);
const generationRecords=fs.readdirSync(P(D+'/sources')).filter(n=>n.endsWith('-result.json')).sort().map(name=>{const file=D+'/sources/'+name,r=json(file),image=file.replace(/-result\.json$/,'.png');assert.equal(r.imageReturned,true);assert.equal(sha(read(image)),sha(fs.readFileSync(r.imagePath)));return{receipt:file,receiptSha256:sha(read(file)),archiveObservedMtime:fs.statSync(P(file)).mtime.toISOString(),observedCompletedAt:r.observedCompletedAt||null,providerTimestampAvailable:false,imagePath:r.imagePath,imageReturned:true,archivedImage:image,imageSha256:sha(read(image))};});
assert.equal(generationRecords.length,57);
fs.writeFileSync(P(D+'/generation-observation-index.json'),JSON.stringify({schemaVersion:1,method:'Actual tool receipts and archive filesystem modification observations. observedCompletedAt is the tool return observation when recorded; filesystem times are not provider generation timestamps. Originals remain preserved.',imageGenerationCalls:57,imageReturns:57,refusalsObserved:0,records:generationRecords},null,2)+'\n');
const exclusions=records.records.map(r=>({heroId:r.heroId,observedDrawings:r.sources.reduce((n,s)=>n+s.nativeDrawingsObserved,0),retainedDrawings:r.sources.reduce((n,s)=>n+s.retainedDrawings,0),excludedDrawings:r.sources.reduce((n,s)=>n+s.excludedDrawings,0),sources:r.sources.map(s=>({path:s.archivedPath,sha256:s.sha256,actualBodies:s.nativeDrawingsObserved,alphaComponents:s.nativeComponentsObserved,retainedBodies:s.retainedDrawings,excludedBodies:s.excludedDrawings,countBasis:s.observedDrawingCountBasis})),correctionNotes:r.correctionNotes}));
assert.ok(exclusions.every(e=>e.retainedDrawings===48));
fs.writeFileSync(P(D+'/source-exclusions.json'),JSON.stringify({schemaVersion:1,selectedCharacters:40,imageGenerationCalls:57,observedBodies:exclusions.reduce((n,e)=>n+e.observedDrawings,0),retainedBodies:1920,excludedBodies:exclusions.reduce((n,e)=>n+e.excludedDrawings,0),incompleteBodiesAndEffectOnlyComponentsDoNotCountAsDelivered:true,entries:exclusions},null,2)+'\n');
const gallery=D+'/final-review';fs.mkdirSync(P(gallery),{recursive:true});
const panels=[];
for(let i=0;i<40;i+=4){const group=records.records.slice(i,i+4),images=[];for(const [col,r]of group.entries()){images.push({input:await sharp(P(D+'/native-contacts/'+r.heroId+'.png')).resize(256,512,{kernel:'nearest'}).toBuffer(),left:col*256,top:32});images.push({input:await sharp(P(D+'/base-contacts/'+r.heroId+'.png')).resize(256,256,{kernel:'nearest'}).toBuffer(),left:col*256,top:544});const label=r.heroId.replaceAll('_',' ').slice(0,38);images.push({input:Buffer.from(`<svg width="256" height="32"><rect width="256" height="32" fill="#152232"/><text x="4" y="18" fill="white" font-size="10">${label}</text></svg>`),left:col*256,top:0});}const file=gallery+'/panel-'+String(i/4+1).padStart(2,'0')+'.png';await sharp({create:{width:1024,height:800,channels:3,background:'#243148'}}).composite(images).png().toFile(P(file));panels.push({file,sha256:sha(read(file)),heroIds:group.map(r=>r.heroId),purpose:'QA only: native32 above and base16 below, four columns per character; scaled pixel contacts are not new authored poses.'});}
fs.writeFileSync(P(gallery+'/index.json'),JSON.stringify({schemaVersion:1,characters:40,panels},null,2)+'\n');
console.log(JSON.stringify({calls:generationRecords.length,observedBodies:exclusions.reduce((n,e)=>n+e.observedDrawings,0),retainedBodies:1920,excludedBodies:exclusions.reduce((n,e)=>n+e.excludedDrawings,0),panels:panels.length}));
