import fs from'node:fs';import path from'node:path';import sharp from'sharp';import{COMBAT_SOURCE_DEFINITIONS_WAVE15_PART_C as SOURCES}from'../../../src/game/combat/combatSourceDefinitionsWave15PartC.js';
const R=path.resolve(import.meta.dirname,'../../..'),D=import.meta.dirname,reg=JSON.parse(fs.readFileSync(path.join(R,'src/game/combat/characterAnimationsWave15PartC.generated.json'))),ids=Object.entries(SOURCES).filter(([,s])=>['projectile','beam'].includes(s.delivery)).map(([id])=>id);fs.mkdirSync(path.join(D,'anchor-review/grids'),{recursive:true});
const overlay=['<svg width="1024" height="512" xmlns="http://www.w3.org/2000/svg">'];for(let r=0;r<2;r++)for(let c=0;c<4;c++){for(let p=32;p<256;p+=32){overlay.push(`<path d="M${c*256+p} ${r*256}v256 M${c*256} ${r*256+p}h256" stroke="#7892ac" opacity=".3" stroke-width="1"/><text x="${c*256+p+2}" y="${r*256+12}" font-size="9" fill="#e1efff">${p}</text><text x="${c*256+2}" y="${r*256+p+9}" font-size="9" fill="#e1efff">${p}</text>`);}overlay.push(`<rect x="${c*256}" y="${r*256}" width="256" height="256" fill="none" stroke="#adc8e5"/><text x="${c*256+5}" y="${r*256+25}" font-size="9" fill="#b6cee9">${r?'super':'fatality'} c${c}</text>`);}overlay.push('</svg>');
for(const id of ids){const e=reg.entries.find(e=>e.heroId===id),bytes=fs.readFileSync(path.join(R,'public'+e.sheet));const a=await sharp(bytes).extract({left:0,top:e.phases.fatality.row*256,width:1024,height:256}).png().toBuffer(),b=await sharp(bytes).extract({left:0,top:e.phases.super.row*256,width:1024,height:256}).png().toBuffer();await sharp({create:{width:1024,height:512,channels:4,background:'#243148'}}).composite([{input:a,left:0,top:0},{input:b,left:0,top:256},{input:Buffer.from(overlay.join('')),left:0,top:0}]).png().toFile(path.join(D,'anchor-review/grids',id+'.png'));}
